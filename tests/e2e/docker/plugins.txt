# Plugins baked into the end-to-end Jenkins image. THE FULL RESOLVED SET,
# with every version pinned — not just the ones we asked for.
#
# Why the whole set. Pinning the base image by digest and naming three plugins
# is not reproducibility: jenkins-plugin-cli resolves transitive dependencies
# to latest-compatible, so the direct asks pulled in ~54 more at whatever versions
# were current that day. Rebuilding the same Dockerfile next month would
# install different ones. Silent version drift is the class of bug this suite
# exists to catch, so it must not be in the suite itself.
#
# To regenerate after changing a direct dependency below, build the image and
# read the versions back out of it:
#
#   cd tests/e2e/docker
#   docker build -t jeeves-e2e-jenkins:local .
#   docker run --rm jeeves-e2e-jenkins:local sh -c \
#     'for f in /usr/share/jenkins/ref/plugins/*.jpi; do \
#        n=$(basename "$f" .jpi); \
#        v=$(unzip -p "$f" META-INF/MANIFEST.MF | tr -d "\r" \
#            | awk -F": " "/^Plugin-Version:/{print \$2}"); \
#        echo "$n:$v"; done' | sort
#
